"""
Courier Tracking System
Frontend: Python Tkinter
Backend: Oracle Database (cx_Oracle)

Single-file project: courier_system.py

Run instructions (short):
1. Install requirements: pip install cx_Oracle
   NOTE: cx_Oracle requires Oracle Instant Client or Oracle client libraries configured on your machine.
2. Update the ORACLE_CONFIG dictionary below with your Oracle username, password and DSN (service name).
   Example DSN: "localhost/orclpdb" or "localhost/XEPDB1" depending on your installation.
3. Run: python courier_system.py

This script will attempt to create the required tables on first run. It creates a default admin account:
    username: admin
    password: admin

Security note: This is a student project example. Passwords are hashed with sha256 but for production use stronger practices.

"""

import tkinter as tk
from tkinter import messagebox, simpledialog
import cx_Oracle
import hashlib
import datetime
import sys
from functools import partial

# -------------------------------
# Configuration - change these
# -------------------------------
ORACLE_CONFIG = {
    'user': 'YOUR_ORACLE_USERNAME',
    'password': 'YOUR_ORACLE_PASSWORD',
    'dsn': 'localhost/orcl'  # change to your service name, e.g. localhost/XEPDB1
}

COST_PER_KG = 50.0  # simple pricing rule

# -------------------------------
# Utility functions
# -------------------------------

def hash_password(password: str) -> str:
    return hashlib.sha256(password.encode('utf-8')).hexdigest()


# -------------------------------
# Database helpers
# -------------------------------
class Database:
    def __init__(self, config):
        try:
            self.conn = cx_Oracle.connect(config['user'], config['password'], config['dsn'])
            self.cursor = self.conn.cursor()
        except cx_Oracle.DatabaseError as e:
            messagebox.showerror("Database Error", f"Could not connect to Oracle DB:\n{e}")
            sys.exit(1)

    def init_db(self):
        # Create tables if they don't exist. We catch errors if tables already exist.
        ddl_users = '''
        CREATE TABLE users (
            user_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            username VARCHAR2(50) UNIQUE NOT NULL,
            password VARCHAR2(256) NOT NULL,
            email VARCHAR2(100),
            phone VARCHAR2(15)
        )
        '''

        ddl_admin = '''
        CREATE TABLE admin (
            admin_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            username VARCHAR2(50) UNIQUE NOT NULL,
            password VARCHAR2(256) NOT NULL
        )
        '''

        ddl_courier = '''
        CREATE TABLE courier (
            courier_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            sender_id NUMBER REFERENCES users(user_id),
            receiver_name VARCHAR2(100) NOT NULL,
            receiver_address VARCHAR2(400) NOT NULL,
            weight NUMBER(7,2),
            cost NUMBER(12,2),
            booking_date DATE DEFAULT SYSDATE,
            status VARCHAR2(50) DEFAULT 'Booked'
        )
        '''

        # Try create
        for ddl in (ddl_users, ddl_admin, ddl_courier):
            try:
                self.cursor.execute(ddl)
                self.conn.commit()
            except cx_Oracle.DatabaseError:
                # Table likely exists
                self.conn.rollback()

        # Ensure a default admin exists
        try:
            self.cursor.execute("SELECT COUNT(*) FROM admin")
            count = self.cursor.fetchone()[0]
            if count == 0:
                hashed = hash_password('admin')
                self.cursor.execute("INSERT INTO admin (username, password) VALUES (:u, :p)", u='admin', p=hashed)
                self.conn.commit()
        except cx_Oracle.DatabaseError:
            self.conn.rollback()

    def register_user(self, username, password, email, phone):
        try:
            hashed = hash_password(password)
            self.cursor.execute("INSERT INTO users (username, password, email, phone) VALUES (:u,:p,:e,:ph)",
                                {'u': username, 'p': hashed, 'e': email, 'ph': phone})
            self.conn.commit()
            return True
        except cx_Oracle.IntegrityError:
            self.conn.rollback()
            return False
        except cx_Oracle.DatabaseError as e:
            self.conn.rollback()
            raise e

    def authenticate_user(self, username, password, role='user'):
        hashed = hash_password(password)
        try:
            if role == 'user':
                self.cursor.execute("SELECT user_id FROM users WHERE username=:u AND password=:p", {'u': username, 'p': hashed})
                row = self.cursor.fetchone()
                return row[0] if row else None
            else:
                self.cursor.execute("SELECT admin_id FROM admin WHERE username=:u AND password=:p", {'u': username, 'p': hashed})
                row = self.cursor.fetchone()
                return row[0] if row else None
        except cx_Oracle.DatabaseError:
            return None

    def book_courier(self, sender_id, receiver_name, receiver_address, weight):
        cost = float(weight) * COST_PER_KG
        try:
            self.cursor.execute(
                "INSERT INTO courier (sender_id, receiver_name, receiver_address, weight, cost) VALUES (:s,:rn,:ra,:w,:c)",
                {'s': sender_id, 'rn': receiver_name, 'ra': receiver_address, 'w': weight, 'c': cost}
            )
            self.conn.commit()
            # fetch latest courier_id
            self.cursor.execute("SELECT courier_id FROM (SELECT courier_id FROM courier ORDER BY courier_id DESC) WHERE ROWNUM=1")
            cid = self.cursor.fetchone()[0]
            return cid
        except cx_Oracle.DatabaseError as e:
            self.conn.rollback()
            raise e

    def get_courier_by_id(self, courier_id):
        try:
            self.cursor.execute("SELECT courier_id, sender_id, receiver_name, receiver_address, weight, cost, booking_date, status FROM courier WHERE courier_id = :id", {'id': courier_id})
            return self.cursor.fetchone()
        except cx_Oracle.DatabaseError:
            return None

    def get_couriers_by_user(self, user_id):
        try:
            self.cursor.execute("SELECT courier_id, receiver_name, receiver_address, weight, cost, booking_date, status FROM courier WHERE sender_id = :s ORDER BY booking_date DESC", {'s': user_id})
            return self.cursor.fetchall()
        except cx_Oracle.DatabaseError:
            return []

    def get_all_couriers(self):
        try:
            self.cursor.execute("SELECT courier_id, sender_id, receiver_name, receiver_address, weight, cost, booking_date, status FROM courier ORDER BY booking_date DESC")
            return self.cursor.fetchall()
        except cx_Oracle.DatabaseError:
            return []

    def get_all_users(self):
        try:
            self.cursor.execute("SELECT user_id, username, email, phone FROM users ORDER BY username")
            return self.cursor.fetchall()
        except cx_Oracle.DatabaseError:
            return []

    def update_courier_status(self, courier_id, new_status):
        try:
            self.cursor.execute("UPDATE courier SET status=:st WHERE courier_id = :id", {'st': new_status, 'id': courier_id})
            self.conn.commit()
            return True
        except cx_Oracle.DatabaseError:
            self.conn.rollback()
            return False

    def delete_courier(self, courier_id):
        try:
            self.cursor.execute("DELETE FROM courier WHERE courier_id = :id", {'id': courier_id})
            self.conn.commit()
            return True
        except cx_Oracle.DatabaseError:
            self.conn.rollback()
            return False

# GUI Application

class App:
    def __init__(self, root, db: Database):
        self.root = root
        self.db = db
        self.current_user_id = None
        self.root.title("Courier Tracking System")
        self.show_login_window()

    # ---------- Login / Register ----------
    def clear_root(self):
        for widget in self.root.winfo_children():
            widget.destroy()

    def show_login_window(self):
        self.clear_root()
        tk.Label(self.root, text="Courier Tracking System", font=("Arial", 18, "bold")).pack(pady=10)

        frame = tk.Frame(self.root)
        frame.pack(pady=10)

        tk.Label(frame, text="Username:").grid(row=0, column=0, sticky='e')
        username_entry = tk.Entry(frame)
        username_entry.grid(row=0, column=1)

        tk.Label(frame, text="Password:").grid(row=1, column=0, sticky='e')
        password_entry = tk.Entry(frame, show='*')
        password_entry.grid(row=1, column=1)

        role_var = tk.StringVar(value='user')
        tk.Radiobutton(frame, text='User', variable=role_var, value='user').grid(row=2, column=0)
        tk.Radiobutton(frame, text='Admin', variable=role_var, value='admin').grid(row=2, column=1)

        def do_login():
            username = username_entry.get().strip()
            password = password_entry.get().strip()
            role = role_var.get()
            if not username or not password:
                messagebox.showwarning("Input needed", "Enter username and password")
                return
            uid = self.db.authenticate_user(username, password, role='admin' if role == 'admin' else 'user')
            if uid:
                if role == 'user':
                    self.current_user_id = uid
                    self.show_user_dashboard()
                else:
                    self.show_admin_dashboard()
            else:
                messagebox.showerror("Login Failed", "Invalid credentials")

        tk.Button(self.root, text="Login", width=20, command=do_login).pack(pady=5)
        tk.Button(self.root, text="Register (New User)", width=20, command=self.show_register_window).pack(pady=5)

    def show_register_window(self):
        self.clear_root()
        tk.Label(self.root, text="Register - New User", font=("Arial", 16)).pack(pady=10)

        frame = tk.Frame(self.root)
        frame.pack(pady=8)

        tk.Label(frame, text="Username:").grid(row=0, column=0, sticky='e')
        username_entry = tk.Entry(frame)
        username_entry.grid(row=0, column=1)

        tk.Label(frame, text="Password:").grid(row=1, column=0, sticky='e')
        password_entry = tk.Entry(frame, show='*')
        password_entry.grid(row=1, column=1)

        tk.Label(frame, text="Email:").grid(row=2, column=0, sticky='e')
        email_entry = tk.Entry(frame)
        email_entry.grid(row=2, column=1)

        tk.Label(frame, text="Phone:").grid(row=3, column=0, sticky='e')
        phone_entry = tk.Entry(frame)
        phone_entry.grid(row=3, column=1)

        def do_register():
            u = username_entry.get().strip()
            p = password_entry.get().strip()
            e = email_entry.get().strip()
            ph = phone_entry.get().strip()
            if not u or not p:
                messagebox.showwarning("Required", "Username and password required")
                return
            ok = self.db.register_user(u, p, e, ph)
            if ok:
                messagebox.showinfo("Registered", "User registered successfully. Please login.")
                self.show_login_window()
            else:
                messagebox.showerror("Error", "Username already exists. Choose another")

        tk.Button(self.root, text="Register", width=20, command=do_register).pack(pady=8)
        tk.Button(self.root, text="Back to Login", width=20, command=self.show_login_window).pack()

    # ---------- User Dashboard ----------
    def show_user_dashboard(self):
        self.clear_root()
        tk.Label(self.root, text=f"User Dashboard", font=("Arial", 16, "bold")).pack(pady=8)
        tk.Label(self.root, text=f"User ID: {self.current_user_id}").pack()

        tk.Button(self.root, text="📦 Book Courier", width=25, command=self.show_book_courier).pack(pady=6)
        tk.Button(self.root, text="🔍 Track Courier", width=25, command=self.show_track_courier).pack(pady=6)
        tk.Button(self.root, text="📋 View My Couriers", width=25, command=self.show_my_couriers).pack(pady=6)
        tk.Button(self.root, text="⚙️ Update Profile (Not implemented)", width=25, command=lambda: messagebox.showinfo("Info","Use DB or re-register to change details"))
        tk.Button(self.root, text="🚪 Logout", width=25, command=self.logout).pack(pady=6)

    def logout(self):
        self.current_user_id = None
        self.show_login_window()

    # ---------- Book Courier ----------
    def show_book_courier(self):
        self.clear_root()
        tk.Label(self.root, text="Book Courier", font=("Arial", 16, "bold")).pack(pady=8)

        frame = tk.Frame(self.root)
        frame.pack(pady=6)

        tk.Label(frame, text="Receiver Name:").grid(row=0, column=0, sticky='e')
        rn = tk.Entry(frame, width=40)
        rn.grid(row=0, column=1)

        tk.Label(frame, text="Receiver Address:").grid(row=1, column=0, sticky='e')
        ra = tk.Entry(frame, width=40)
        ra.grid(row=1, column=1)

        tk.Label(frame, text="Weight (kg):").grid(row=2, column=0, sticky='e')
        wt = tk.Entry(frame)
        wt.grid(row=2, column=1)

        def do_book():
            receiver_name = rn.get().strip()
            receiver_address = ra.get().strip()
            weight = wt.get().strip()
            if not receiver_name or not receiver_address or not weight:
                messagebox.showwarning("Required", "Please fill all fields")
                return
            try:
                weight_val = float(weight)
            except ValueError:
                messagebox.showwarning("Invalid", "Weight must be a number")
                return
            cid = self.db.book_courier(self.current_user_id, receiver_name, receiver_address, weight_val)
            messagebox.showinfo("Booked", f"Courier booked successfully. Tracking ID: {cid}")
            self.show_user_dashboard()

        tk.Button(self.root, text="Book", width=20, command=do_book).pack(pady=8)
        tk.Button(self.root, text="Back", width=20, command=self.show_user_dashboard).pack()

    # ---------- Track Courier ----------
    def show_track_courier(self):
        self.clear_root()
        tk.Label(self.root, text="Track Courier", font=("Arial", 16, "bold")).pack(pady=8)

        frame = tk.Frame(self.root)
        frame.pack(pady=6)

        tk.Label(frame, text="Enter Tracking ID:").grid(row=0, column=0, sticky='e')
        tid = tk.Entry(frame)
        tid.grid(row=0, column=1)

        output = tk.Text(self.root, height=10, width=70)
        output.pack(pady=8)

        def do_track():
            val = tid.get().strip()
            if not val:
                messagebox.showwarning("Required", "Enter a tracking ID")
                return
            try:
                cid = int(val)
            except ValueError:
                messagebox.showwarning("Invalid", "Tracking ID must be numeric")
                return
            row = self.db.get_courier_by_id(cid)
            output.delete('1.0', tk.END)
            if row:
                (courier_id, sender_id, receiver_name, receiver_address, weight, cost, booking_date, status) = row
                booking_date = booking_date.strftime('%Y-%m-%d %H:%M:%S') if isinstance(booking_date, datetime.datetime) else str(booking_date)
                out = f"Tracking ID: {courier_id}\nSender ID: {sender_id}\nReceiver: {receiver_name}\nAddress: {receiver_address}\nWeight: {weight} kg\nCost: ₹{cost}\nBooked on: {booking_date}\nStatus: {status}"
                output.insert(tk.END, out)
            else:
                output.insert(tk.END, "No courier found with that ID")

        tk.Button(self.root, text="Track", width=20, command=do_track).pack()
        tk.Button(self.root, text="Back", width=20, command=self.show_user_dashboard).pack(pady=6)

    # ---------- View My Couriers ----------
    def show_my_couriers(self):
        self.clear_root()
        tk.Label(self.root, text="My Couriers", font=("Arial", 16, "bold")).pack(pady=8)

        text = tk.Text(self.root, width=90, height=20)
        text.pack(pady=6)

        rows = self.db.get_couriers_by_user(self.current_user_id)
        if not rows:
            text.insert(tk.END, "No bookings yet.")
        else:
            for r in rows:
                courier_id, receiver_name, receiver_address, weight, cost, booking_date, status = r
                booking_date = booking_date.strftime('%Y-%m-%d %H:%M:%S') if isinstance(booking_date, datetime.datetime) else str(booking_date)
                text.insert(tk.END, f"ID: {courier_id} | Receiver: {receiver_name} | Addr: {receiver_address} | Weight: {weight} kg | Cost: ₹{cost} | Date: {booking_date} | Status: {status}\n\n")

        tk.Button(self.root, text="Back", width=20, command=self.show_user_dashboard).pack(pady=6)

    # ---------- Admin Dashboard ----------
    def show_admin_dashboard(self):
        self.clear_root()
        tk.Label(self.root, text="Admin Dashboard", font=("Arial", 16, "bold")).pack(pady=8)

        tk.Button(self.root, text="View All Couriers", width=25, command=self.show_all_couriers).pack(pady=6)
        tk.Button(self.root, text="Update Courier Status", width=25, command=self.show_update_status).pack(pady=6)
        tk.Button(self.root, text="View Users", width=25, command=self.show_users).pack(pady=6)
        tk.Button(self.root, text="Delete Courier Record", width=25, command=self.show_delete_courier).pack(pady=6)
        tk.Button(self.root, text="Logout", width=25, command=self.show_login_window).pack(pady=6)

    def show_all_couriers(self):
        self.clear_root()
        tk.Label(self.root, text="All Couriers", font=("Arial", 16, "bold")).pack(pady=8)
        text = tk.Text(self.root, width=120, height=25)
        text.pack(pady=6)
        rows = self.db.get_all_couriers()
        if not rows:
            text.insert(tk.END, "No courier records.")
        else:
            for r in rows:
                courier_id, sender_id, receiver_name, receiver_address, weight, cost, booking_date, status = r
                booking_date = booking_date.strftime('%Y-%m-%d %H:%M:%S') if isinstance(booking_date, datetime.datetime) else str(booking_date)
                text.insert(tk.END, f"ID: {courier_id} | Sender: {sender_id} | Receiver: {receiver_name} | Addr: {receiver_address} | Weight: {weight} kg | Cost: ₹{cost} | Date: {booking_date} | Status: {status}\n\n")
        tk.Button(self.root, text="Back", width=20, command=self.show_admin_dashboard).pack(pady=6)

    def show_update_status(self):
        self.clear_root()
        tk.Label(self.root, text="Update Courier Status", font=("Arial", 16, "bold")).pack(pady=8)

        frame = tk.Frame(self.root)
        frame.pack(pady=6)
        tk.Label(frame, text="Courier ID:").grid(row=0, column=0, sticky='e')
        cid_entry = tk.Entry(frame)
        cid_entry.grid(row=0, column=1)
        tk.Label(frame, text="New Status:").grid(row=1, column=0, sticky='e')
        status_entry = tk.Entry(frame)
        status_entry.grid(row=1, column=1)

        def do_update():
            val = cid_entry.get().strip()
            st = status_entry.get().strip()
            if not val or not st:
                messagebox.showwarning("Required", "Please enter courier ID and new status")
                return
            try:
                cid = int(val)
            except ValueError:
                messagebox.showwarning("Invalid", "Courier ID must be numeric")
                return
            ok = self.db.update_courier_status(cid, st)
            if ok:
                messagebox.showinfo("Updated", "Status updated successfully")
            else:
                messagebox.showerror("Error", "Could not update status")

        tk.Button(self.root, text="Update", width=20, command=do_update).pack(pady=6)
        tk.Button(self.root, text="Back", width=20, command=self.show_admin_dashboard).pack()

    def show_users(self):
        self.clear_root()
        tk.Label(self.root, text="Registered Users", font=("Arial", 16, "bold")).pack(pady=8)
        text = tk.Text(self.root, width=80, height=20)
        text.pack(pady=6)
        rows = self.db.get_all_users()
        if not rows:
            text.insert(tk.END, "No users registered.")
        else:
            for r in rows:
                user_id, username, email, phone = r
                text.insert(tk.END, f"ID: {user_id} | Username: {username} | Email: {email} | Phone: {phone}\n")
        tk.Button(self.root, text="Back", width=20, command=self.show_admin_dashboard).pack(pady=6)

    def show_delete_courier(self):
        self.clear_root()
        tk.Label(self.root, text="Delete Courier Record", font=("Arial", 16, "bold")).pack(pady=8)

        frame = tk.Frame(self.root)
        frame.pack(pady=6)
        tk.Label(frame, text="Courier ID:").grid(row=0, column=0, sticky='e')
        cid_entry = tk.Entry(frame)
        cid_entry.grid(row=0, column=1)

        def do_delete():
            val = cid_entry.get().strip()
            if not val:
                messagebox.showwarning("Required", "Enter courier ID")
                return
            try:
                cid = int(val)
            except ValueError:
                messagebox.showwarning("Invalid", "Courier ID must be numeric")
                return
            if messagebox.askyesno("Confirm", f"Delete courier {cid}? THIS CANNOT BE UNDONE"):
                ok = self.db.delete_courier(cid)
                if ok:
                    messagebox.showinfo("Deleted", "Courier deleted")
                else:
                    messagebox.showerror("Error", "Could not delete courier")

        tk.Button(self.root, text="Delete", width=20, command=do_delete).pack(pady=6)
        tk.Button(self.root, text="Back", width=20, command=self.show_admin_dashboard).pack()


# -------------------------------
# Main
# -------------------------------

def main():
    root = tk.Tk()
    # connect to DB
    db = Database(ORACLE_CONFIG)
    db.init_db()
    app = App(root, db)
    root.geometry('800x600')
    root.mainloop()


if __name__ == '__main__':
    main()
